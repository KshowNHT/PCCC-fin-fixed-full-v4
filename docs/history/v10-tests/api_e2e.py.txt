import io
import json
import os

import urllib.error
import urllib.request

sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))
from tests.test_appraisal_scenarios import CASES  # noqa: E402

API = os.environ.get("API", "http://127.0.0.1:9910")

errors = []


def call(method, path, body=None, token=None, raw=False):
    req = urllib.request.Request(API + path, method=method, data=json.dumps(body).encode() if body is not None else None)
    req.add_header("Content-Type", "application/json")
    if token:
        req.add_header("Authorization", f"Bearer {token}")
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            data = r.read()
            return r.status, (data if raw else json.loads(data or b"{}"))
    except urllib.error.HTTPError as e:
        data = e.read()
        try:
            return e.code, json.loads(data)
        except Exception:  # noqa: BLE001
            return e.code, data


def check(cond, msg):
    print(("   ok  : " if cond else "   FAIL: ") + msg)
    if not cond:
        errors.append(msg)


status, tok = call("POST", "/api/auth/login-json", {"username": USER, "password": PASS})
assert status == 200, tok
T = tok["access_token"]

print(f"### 1. {len(CASES)} kịch bản qua PUT /api/sessions/{{id}}/project")
FIELDS = ("type", "floors", "totalFloorArea", "floorArea", "pcccHeight", "height", "fireRating", "commercialDetails", "floorFunctions")
for cid, desc, project, exp in CASES:
    if cid == "C29":
        continue  # chuỗi OCR: kiểm tra riêng ở mục 5 (API phải chặn 422)
    s, sess = call("POST", "/api/sessions", {}, T)
    sid = sess["id"]
    body = {k: project[k] for k in FIELDS if k in project and project[k] is not None}
    s, res = call("PUT", f"/api/sessions/{sid}/project?appraise=false", body, T)
    if s != 200:
        check(False, f"{cid}: API trả {s} {str(res)[:120]}")
        continue
    b = res["appraisalBasis"]
    by = {c["stt"]: c["result"] for c in res["checklist"]}
    # Hồ sơ mới qua API luôn mang loại công trình mặc định (nhà ở kết hợp kinh doanh),
    # nên ca "hồ sơ rỗng" (C28) thuộc nhóm đó thay vì "chưa xác định" như khi gọi engine với dict rỗng.
    exp_category = "mixed_residential" if cid == "C28" else exp["category"]
    ok = (b["status"] == exp["status"] and b["category"] == exp_category and set(b["triggeredBy"]) == exp["triggered"]
          and all(by[k] == v for k, v in exp.get("items", {}).items())
          and res["isSubjectToAppraisal"] == (exp["status"] == "required")
          and res["appraisalReason"] == b["summary"])
    check(ok, f"{cid} {desc[:70]}")

print("\n### 4. File Word xuất ra")
from docx import Document  # noqa: E402


def docx_text(sid_):
    s_, data = call("POST", "/api/report/export", {"sessionId": sid_}, T, raw=True)
    assert s_ == 200
    d = Document(io.BytesIO(data))
    parts = [p.text for p in d.paragraphs]
    for t in d.tables:
        for row in t.rows:
            parts += [c.text for c in row.cells]
    return "\n".join(parts)
