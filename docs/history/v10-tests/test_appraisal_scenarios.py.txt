_NEG = re.compile(r"(không|chưa)\s+(thuộc diện\s+)?(bắt buộc|cưỡng bức)|dưới ngưỡng bắt buộc|không đủ để khẳng định")


def check_invariants(project, ev, label):
    errs = []
    basis = ev["appraisalBasis"]
    cl = ev["checklist"]
    status = basis["status"]

    # G1
    if [c["stt"] for c in cl] != list(range(1, 18)):
        errs.append("G1: checklist không đủ 17 mục theo thứ tự")
    # G2
    mapping = {"required": "Không đạt", "not_required": "Không bắt buộc", "insufficient_data": "Cần xem xét"}
    if cl[0]["result"] != mapping[status]:
        errs.append(f"G2: mục 1 = {cl[0]['result']} nhưng status = {status}")
    if ev["isSubjectToAppraisal"] != (status == "required"):
        errs.append("G2b: isSubjectToAppraisal lệch status")
    # G3
    if status == "required" and "BẮT BUỘC thẩm duyệt" not in basis["summary"] + basis["conclusion"]:
        errs.append("G3: required nhưng lý do không nói bắt buộc")
    if status != "required" and re.search(r"BẮT BUỘC thẩm duyệt", basis["summary"] + basis["conclusion"]):
        errs.append("G3: không phải required nhưng lý do nói BẮT BUỘC thẩm duyệt")
    # G4: "ĐẠT HOẶC VƯỢT" chỉ khi có ngưỡng đã đạt
    met_rows = [r for r in basis["thresholds"] if r["state"] == "met"]
    if "ĐẠT HOẶC VƯỢT" in basis["conclusion"] and not met_rows:
        errs.append("G4: câu 'ĐẠT HOẶC VƯỢT' xuất hiện khi không có ngưỡng nào đạt  <-- LỖI TRONG ẢNH")
    if status == "required" and basis["ruleMode"] == "threshold" and "ĐẠT HOẶC VƯỢT" not in basis["conclusion"]:
        errs.append("G4b: required theo ngưỡng nhưng kết luận không nêu ngưỡng đạt")
    # G5/G6: ngưỡng tính độc lập
    vals = {"floors": to_number(project.get("floors")), "totalFloorArea": to_number(project.get("totalFloorArea")),
            "height": effective_height(project)}
    info = CATEGORY_INFO[basis["category"]]
    if len(basis["thresholds"]) != len(info["thresholds"]):
        errs.append("G5: số ngưỡng lệch danh mục")
    for row in basis["thresholds"]:
        actual = vals[row["key"]]
        exp = "missing" if actual <= 0 else ("met" if actual >= row["limit"] else "not_met")
        if row["state"] != exp:
            errs.append(f"G5: ngưỡng {row['key']} state={row['state']} kỳ vọng {exp} (actual={actual}, limit={row['limit']})")
    if set(basis["triggeredBy"]) != {r["key"] for r in met_rows}:
        errs.append("G6: triggeredBy lệch")
    expected_required = info["rule"] == "always" or bool(met_rows)
    if (status == "required") != expected_required:
        errs.append("G6b: status required không khớp quy tắc ngưỡng")
    # G7: số liệu thực xuất hiện trong lý do
    text = basis["summary"] + " " + basis["conclusion"]
    if status in ("required", "not_required") and info["rule"] == "threshold":
        rows_must_show = met_rows if status == "required" else basis["thresholds"]
        for row in rows_must_show:
            if row["actualText"] not in text:
                errs.append(f"G7: lý do thiếu giá trị thực '{row['actualText']}' của {row['key']}")
    # G8: note-vs-result
    for c in cl:
        note, res, stt = c["note"], c["result"], c["stt"]
        low = _NEG.sub("", note.lower())
        if res == "Không bắt buộc" and "bắt buộc" in low:
            errs.append(f"G8: mục {stt} 'Không bắt buộc' nhưng note nói bắt buộc")
        if res in ("Khuyến nghị", "Khuyến nghị mạnh") and re.search(r"bắt buộc|yêu cầu|\bphải\b", low):
            errs.append(f"G8: mục {stt} '{res}' nhưng note dùng ngôn từ bắt buộc")
        if res == "Không đạt" and re.search(r"không bắt buộc", note.lower()):
            errs.append(f"G8: mục {stt} 'Không đạt' nhưng note nói không bắt buộc")
        if res == "Cần xem xét" and stt in (1, 2, 9, 10, 11, 12):
            if not re.search(r"chưa đủ|chưa chọn|cần đối chiếu", note.lower()):
                errs.append(f"G8: mục {stt} 'Cần xem xét' không nêu lý do thiếu dữ liệu")
    # G9/G10
    if json.dumps(ev, ensure_ascii=False, sort_keys=True) != json.dumps(evaluate_appraisal(project), ensure_ascii=False, sort_keys=True):
        errs.append("G9: không tái lập")
    # G11
    fl, fa, ta = vals["floors"], to_number(project.get("floorArea")), vals["totalFloorArea"]
    mismatch = fl > 0 and fa > 0 and abs(ta - fa * fl) > 10
    if mismatch != (AREA_MISMATCH_WARNING in ev["warnings"]):
        errs.append("G11: cảnh báo lệch diện tích không khớp công thức")
    # G13 không có chuỗi rác
    for c in cl:
        if re.search(r"None|nan|undefined|\{|\}", c["note"]):
            errs.append(f"G13: mục {c['stt']} chứa chuỗi rác: {c['note'][:60]}")
    if re.search(r"None|nan|undefined|\{|\}", basis["summary"] + basis["conclusion"]):
        errs.append("G13: lý do chứa chuỗi rác")
    return [f"[{label}] {e}" for e in errs]


def run_case(case):
    cid, desc, project, exp = case
    before = copy.deepcopy(project)
    ev = evaluate_appraisal(project)
    errs = []
    if project != before:
        errs.append(f"[{cid}] G12: evaluate_appraisal làm thay đổi dữ liệu đầu vào")
    errs += check_invariants(project, ev, cid)
    b = ev["appraisalBasis"]
    if b["status"] != exp["status"]:
        errs.append(f"[{cid}] status = {b['status']}, kỳ vọng {exp['status']}")
    if b["category"] != exp["category"]:
        errs.append(f"[{cid}] category = {b['category']}, kỳ vọng {exp['category']}")
    if set(b["triggeredBy"]) != exp["triggered"]:
        errs.append(f"[{cid}] triggeredBy = {b['triggeredBy']}, kỳ vọng {sorted(exp['triggered'])}")
    by_stt = {c["stt"]: c["result"] for c in ev["checklist"]}
    for stt, want in exp.get("items", {}).items():
        if by_stt[stt] != want:
            errs.append(f"[{cid}] mục {stt} = '{by_stt[stt]}', kỳ vọng '{want}'")
    if "compliant" in exp:
        got = sum(1 for r in by_stt.values() if r in ("Đạt", "Không bắt buộc"))
        if got != exp["compliant"]:
            errs.append(f"[{cid}] số mục 'Đạt & Không bắt buộc' = {got}, kỳ vọng {exp['compliant']}")
    if exp.get("no_thresholds") and b["thresholds"]:
        errs.append(f"[{cid}] nhóm không phụ thuộc quy mô nhưng vẫn hiện ngưỡng")
    if "warning_contains" in exp and not any(exp["warning_contains"] in w for w in ev["warnings"]):
        errs.append(f"[{cid}] thiếu cảnh báo chứa '{exp['warning_contains']}'")
    if "no_warning_contains" in exp and any(exp["no_warning_contains"] in w for w in ev["warnings"]):
        errs.append(f"[{cid}] có cảnh báo không mong muốn '{exp['no_warning_contains']}'")
    if "missing_contains" in exp and exp["missing_contains"] not in b["missingFields"]:
        errs.append(f"[{cid}] missingFields = {b['missingFields']} thiếu '{exp['missing_contains']}'")
    if "evidence_contains" in exp and exp["evidence_contains"] not in b["categoryEvidence"]:
        errs.append(f"[{cid}] categoryEvidence = '{b['categoryEvidence']}' thiếu '{exp['evidence_contains']}'")
    if "scale_text" in exp and b["actualScale"] != exp["scale_text"]:
        errs.append(f"[{cid}] actualScale = '{b['actualScale']}', kỳ vọng '{exp['scale_text']}'")
    if exp.get("area_mismatch") and AREA_MISMATCH_WARNING not in ev["warnings"]:
        errs.append(f"[{cid}] thiếu cảnh báo lệch diện tích")
    return errs


def fuzz(n=3000, seed=20241):
    rnd = random.Random(seed)
    types = list(CATEGORY_INFO.keys()) + [
        MIXED, "Nhà ở riêng lẻ", KARAOKE, "Gara ô tô", "Văn phòng, khách sạn công cộng", "Nhà kho xếp giá đỡ kệ cao",
        "Chợ, Trung tâm thương mại", "Nhà máy, nhà xưởng sản xuất", "Cơ sở giáo dục, trường học",
        "Rạp hát, hội trường, trung tâm tổ chức sự kiện", "", None, "Linh tinh 123", "karaoke", "nhà kho", "trường mầm non",
    ]
    nums = [None, 0, "", 1, 2, 3, 4, 5, 6, 7, 8, 12, 25, 99, 100, 499.5, 999, 1000, 1499, 1500, 2999, 3000, 3001, 5000, 12000, "7 tầng", "3.000", "12,5", "abc", -3, float("nan")]
    ratings = ["Bậc I", "Bậc II", "Bậc III", "Bậc IV", "Bậc V", "Chưa lựa chọn", "", None]
    details = ["", None, "karaoke", "gần chợ", "đối diện trường học", "kho hàng", "cửa hàng discount", "Tầng 1 văn phòng"]
    funcs = ["Nhà ở riêng lẻ / Dân dụng", "Dịch vụ giải trí Karaoke, Vũ trường", "Nhà kho lưu trữ hàng hóa", None, 5]
    fails = []
    for i in range(n):
        proj = {"type": rnd.choice(types), "floors": rnd.choice(nums), "totalFloorArea": rnd.choice(nums),
                "floorArea": rnd.choice(nums), "pcccHeight": rnd.choice(nums), "height": rnd.choice(nums),
                "fireRating": rnd.choice(ratings), "commercialDetails": rnd.choice(details),
                "floorFunctions": [rnd.choice(funcs) for _ in range(rnd.randint(0, 6))]}
        try:
            ev = evaluate_appraisal(proj)
            fails += check_invariants(proj, ev, f"fuzz#{i}") if False else [f"{e} | input={proj}" for e in check_invariants(proj, ev, f"fuzz#{i}")]
        except Exception as exc:  # noqa: BLE001
            fails.append(f"[fuzz#{i}] NGOẠI LỆ {type(exc).__name__}: {exc} | input={proj}")
        if len(fails) > 15:
            break
    return fails
