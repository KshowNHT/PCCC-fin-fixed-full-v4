def run_A():
    errs = []
    for cid, desc, extracted, current, text, exp in A_CASES:
        r = sanitize_extracted_info(extracted, current, text)
        if "accepted" in exp and r["accepted"] != exp["accepted"]:
            errs.append(f"[{cid}] {desc}: accepted = {r['accepted']}, kỳ vọng {exp['accepted']}")
        if "rejected_n" in exp and len(r["rejected"]) != exp["rejected_n"]:
            errs.append(f"[{cid}] {desc}: số mục bị từ chối = {len(r['rejected'])}, kỳ vọng {exp['rejected_n']} ({r['rejected']})")
        if "derived_n" in exp and len(r["derived"]) != exp["derived_n"]:
            errs.append(f"[{cid}] {desc}: derived = {len(r['derived'])}, kỳ vọng {exp['derived_n']}")
        if exp.get("check_valid"):
            merged = {k: v for k, v in {**{f: current[f] for f in ("floors", "floorArea", "totalFloorArea", "pcccHeight") if to_number(current.get(f)) > 0}, **{k: v for k, v in r["accepted"].items() if k in ("floors", "floorArea", "totalFloorArea", "pcccHeight")}}.items()}
            try:
                ProjectUpdateRequest(**merged)
            except Exception as exc:  # noqa: BLE001
                errs.append(f"[{cid}] {desc}: sau khi lọc vẫn vi phạm luật dữ liệu: {exc}")
        print(("PASS " if not any(e.startswith(f"[{cid}]") for e in errs) else "FAIL ") + f"{cid:5s} {desc}")
    return errs


def run_B():
    errs = []
    for cid, desc, project, reply, want_violation, must_keep, must_drop in B_CASES:
        basis = evaluate_appraisal(project)["appraisalBasis"]
        out, violations = guard_reply(reply, basis, project)
        bad = []
        if bool(violations) != want_violation:
            bad.append(f"vi phạm={bool(violations)}, kỳ vọng {want_violation} ({violations})")
        body = out.split("ℹ️ Hệ thống đã lược bỏ")[0]
        for k in must_keep:
            if k not in out:
                bad.append(f"thiếu '{k}' trong kết quả")
        for k in must_drop:
            if k.lower() in body.lower():
                bad.append(f"vẫn còn '{k}' trong phần giữ lại")
        errs += [f"[{cid}] {desc}: {b}" for b in bad]
        print(("PASS " if not bad else "FAIL ") + f"{cid:5s} {desc}")
    return errs


def run_C(n=4000, seed=7):
    rnd = random.Random(seed)
    errs = []
    nums = [None, 0, -1, 1, 2, 3, 5, 7, 8, 12, 18, 25, 100, 500, 900, 1000, 3000, 9999, 5.5, "5", "500", "abc", float("nan"), 1e12]
    texts = [None, "", "nhà 5 tầng 500 m2", "bậc II karaoke", "tầng 1 cửa hàng, 3 tầng, 1.200 m2 cao 12,5m", "mười hai tầng", "bậc 3"]
    types = [None, "", KARAOKE, "Quán karaoke", "Kho hàng", "xyz", MIXED, "Chợ", 5]
    for i in range(n):
        current = {"type": rnd.choice([MIXED, KARAOKE, "", None]), "floors": rnd.choice([0, 3, 5, 8]), "totalFloorArea": rnd.choice([0, 500, 3000]),
                   "floorArea": rnd.choice([0, 100]), "pcccHeight": rnd.choice([0, 18]), "fireRating": rnd.choice(["Bậc III", "Chưa lựa chọn", "Bậc II"]),
                   "name": rnd.choice(["Dự án mới", "Nhà A", ""]), "commercialDetails": rnd.choice(["", "tạp hóa"]), "floorFunctions": []}
        ai = {"floors": rnd.choice(nums), "totalFloorArea": rnd.choice(nums), "floorArea": rnd.choice(nums), "pcccHeight": rnd.choice(nums),
              "height": rnd.choice(nums), "basements": rnd.choice(nums), "type": rnd.choice(types), "fireRating": rnd.choice([None, "Bậc II", "Bậc VI", "x"]),
              "name": rnd.choice([None, "Tên AI"]), "commercialDetails": rnd.choice([None, "karaoke tầng 1", "kho"]),
              "floorFunctions": rnd.choice([None, ["Karaoke"], ["Nhà ở", "Cửa hàng"], ["???"], [1, 2]]), "ghost": 1}
        text = rnd.choice(texts)
        try:
            r = sanitize_extracted_info(ai, current, text)
        except Exception as exc:  # noqa: BLE001
            errs.append(f"[fuzzA#{i}] NGOẠI LỆ {type(exc).__name__}: {exc} | ai={ai}")
            continue
        acc = r["accepted"]
        known = {"floors", "totalFloorArea", "floorArea", "pcccHeight", "height", "basements", "length", "width", "type", "fireRating", "name",
                 "location", "investor", "designer", "stage", "commercialDetails", "floorFunctions"}
        if not set(acc) <= known:
            errs.append(f"[fuzzA#{i}] khoá lạ lọt vào accepted: {set(acc) - known}")
        base_valid = True
        try:
            ProjectUpdateRequest(**{f: current[f] for f in ("floors", "floorArea", "totalFloorArea", "pcccHeight") if to_number(current.get(f)) > 0})
        except Exception:  # noqa: BLE001
            base_valid = False
        if base_valid:
            merged = {f: current[f] for f in ("floors", "floorArea", "totalFloorArea", "pcccHeight") if to_number(current.get(f)) > 0}
            merged.update({k: v for k, v in acc.items() if k in ("floors", "floorArea", "totalFloorArea", "pcccHeight", "height", "basements", "length", "width")})
            try:
                ProjectUpdateRequest(**merged)
            except Exception as exc:  # noqa: BLE001
                errs.append(f"[fuzzA#{i}] hồ sơ sau khi nhận số AI vi phạm luật dữ liệu: {exc} | acc={acc} cur={current}")
        if text is not None:
            ns = numbers_in_text(text)
            for f in ("floors", "floorArea", "pcccHeight", "height", "basements", "length", "width"):
                if f in acc and not any(abs(float(acc[f]) - x) < 1e-6 for x in ns):
                    errs.append(f"[fuzzA#{i}] {f}={acc[f]} được nhận nhưng không có trong lời người dùng '{text}'")
            if "totalFloorArea" in acc and not any(abs(acc["totalFloorArea"] - x) < 1e-6 for x in ns):
                fl = acc.get("floors", current["floors"])
                fa = acc.get("floorArea", current["floorArea"])
                if not (fl and fa and abs(acc["totalFloorArea"] - fl * fa) < 1e-6):
                    errs.append(f"[fuzzA#{i}] totalFloorArea={acc['totalFloorArea']} không có căn cứ")
        for f in ("name",):
            if f in acc and str(current.get(f) or "").strip().lower() not in ("", "dự án mới"):
                errs.append(f"[fuzzA#{i}] ghi đè {f} đã có")
        if "commercialDetails" in acc and str(current.get("commercialDetails") or "").strip():
            errs.append(f"[fuzzA#{i}] ghi đè commercialDetails đã có")
        if "fireRating" in acc and acc["fireRating"] not in KNOWN_RATINGS:
            errs.append(f"[fuzzA#{i}] fireRating lạ {acc['fireRating']}")
        if "floorFunctions" in acc and not set(acc["floorFunctions"]) <= ALLOWED_FLOOR_FUNCTIONS:
            errs.append(f"[fuzzA#{i}] floorFunctions ngoài danh sách")
        if "type" in acc:
            from app.appraisal_basis import classify_building
            if classify_building({"type": acc["type"]})[0] == "unclassified":
                errs.append(f"[fuzzA#{i}] type nhận về nhóm chưa xác định: {acc['type']}")
        if text is not None and acc.get("floorFunctions") and any("Karaoke" in f for f in acc["floorFunctions"]) and not ("karaoke" in text.lower()):
            errs.append(f"[fuzzA#{i}] nhận tầng karaoke mà lời người dùng không nhắc")
        if text is not None and "type" in acc and "Karaoke" in acc["type"] and current.get("type") != KARAOKE and not ("karaoke" in text.lower()):
            errs.append(f"[fuzzA#{i}] đổi sang karaoke mà lời người dùng không nhắc")
        if len(errs) > 12:
            break

    # --- hệ thống không được tự bắt lỗi chính câu chữ của mình ---
    types2 = [MIXED, KARAOKE, "Gara ô tô", "Nhà kho xếp giá đỡ kệ cao", "Chợ, Trung tâm thương mại", "Cơ sở giáo dục, trường học", "", "Công trình khác"]
    for i in range(n):
        proj = {"type": rnd.choice(types2), "floors": rnd.choice([0, 1, 2, 3, 4, 5, 6, 7, 8, 12]), "totalFloorArea": rnd.choice([0, 100, 500, 999, 1000, 1500, 2999, 3000, 9000]),
                "floorArea": rnd.choice([0, 100]), "pcccHeight": rnd.choice([0, 8, 18, 25, 31]), "fireRating": rnd.choice(["Bậc II", "Bậc III", ""])}
        ev = evaluate_appraisal(proj)
        basis = ev["appraisalBasis"]
        own = [basis["summary"], basis["conclusion"], basis["legalText"]] + [c["note"] for c in ev["checklist"]] + ev["warnings"]
        for piece in own:
            out, v = guard_reply(piece, basis, proj)
            if v:
                errs.append(f"[fuzzB#{i}] hệ thống tự bắt lỗi chính câu chữ của mình: {v} | '{piece[:90]}' | proj={proj}")
                break
        kept, dropped = guard_warnings(ev["warnings"], basis, proj)
        if dropped:
            errs.append(f"[fuzzB#{i}] guard_warnings loại cảnh báo của chính hệ thống | proj={proj}")
        if len(errs) > 12:
            break
    print(("PASS " if not errs else "FAIL ") + f"fuzz tính chất ({n} đầu vào AI + {n} hồ sơ tự kiểm tra)")
    return errs
